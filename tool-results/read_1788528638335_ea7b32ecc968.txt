    41→    41→function isPlainObject(v: unknown): v is Record<string, unknown> {
    42→    42→  return typeof v === 'object' && v !== null && !Array.isArray(v);
    43→    43→}
    44→    44→
    45→    45→function storageKey(base: string, version: string): string {
    46→    46→  return `${base}@${version}`;
    47→    47→}
    48→    48→
    49→    49→/** Canonical deterministic order for contract metas: (base, version). */
    50→    50→function compareMeta(a: ContractMeta, b: ContractMeta): number {
    51→    51→  if (a.base !== b.base) return a.base < b.base ? -1 : 1;
    52→    52→  return compareVersions(a.version, b.version);
    53→    53→}
    54→    54→
    55→    55→/** Validate an untrusted parsed meta record; `'corrupt'` on any deviation. */
    56→    56→function assertMetaShape(value: unknown, file: string): ContractMeta {
    57→    57→  if (!isPlainObject(value)) {
    58→    58→    throw new RegistryError('corrupt', `${file}: expected a JSON object`);
    59→    59→  }
    60→    60→  for (const field of ['packageName', 'base', 'version', 'hash', 'shortHash'] as const) {
    61→    61→    if (typeof value[field] !== 'string') {
    62→    62→      throw new RegistryError('corrupt', `${file}: field '${field}' must be a string`);
    63→    63→    }
    64→    64→  }
    65→    65→  if (!Array.isArray(value.imports) || !value.imports.every((x) => typeof x === 'string')) {
    66→    66→    throw new RegistryError('corrupt', `${file}: field 'imports' must be an array of strings`);
    67→    67→  }
    68→    68→  assertSha256Hex(value.hash, `${file}: meta.hash`);
    69→    69→  return value as unknown as ContractMeta;
    70→    70→}
    71→    71→
    72→    72→/** Validate an untrusted parsed pointer record; `'corrupt'` on any deviation. */
    73→    73→function assertPointerShape(value: unknown, file: string): PointerFile {
    74→    74→  if (!isPlainObject(value)) {
    75→    75→    throw new RegistryError('corrupt', `${file}: expected a JSON object`);
    76→    76→  }
    77→    77→  const hash = assertSha256Hex(value.hash, `${file}: pointer.hash`);
    78→    78→  if (typeof value.package !== 'string' || !isValidPackageName(value.package)) {
    79→    79→    throw new RegistryError(
    80→    80→      'corrupt',
    81→    81→      `${file}: pointer.package ${JSON.stringify(value.package)} is not a valid package name`,
    82→    82→    );
    83→    83→  }
    84→    84→  return { hash, package: value.package };
    85→    85→}

    87→    87→/**
    88→    88→ * Local, content-addressed registry of immutable contract versions.
    89→    89→ *
    90→    90→ * Storage layout (all writes atomic: hidden same-directory tmp file +
    91→    91→ * rename; all JSON deterministic — sorted keys via `canonicalJson`):
    92→    92→ *
    93→    93→ * ```
    94→    94→ * <root>/
    95→    95→ *   objects/<hash[0:2]>/<hash>.json          ← canonicalJson(ir), content-addressed
    96→    96→ *   packages/<base>/<version>/contract.json  ← { "hash": ..., "package": <name> }
    97→    97→ *   packages/<base>/<version>/meta.json      ← ContractMeta
    98→    98→ *   index.json                               ← { "contracts": [ContractMeta...] }
    99→    99→ * ```
   100→   100→ *
   101→   101→ * Key semantics:
   102→   102→ * - **Content addressing.** A version's identity is `hashPackage(ir)`.
   103→   103→ *   Republishing identical content is a no-op returning the original meta;
   104→   104→ *   republishing different content under an existing base/version throws
   105→   105→ *   `'immutable'` — published versions are never mutated.
   106→   106→ * - **Lookups are keyed by `(base, version)`.** The base is the package name
   107→   107→ *   minus a version-shaped final segment (`'payments.v1'` and `'payments'`
   108→   108→ *   both address the base `'payments'`).
   109→   109→ * - **Path safety.** Every name/version/hash that reaches a filesystem path
   110→   110→ *   is validated first, whether it came from a caller or from stored data.
   111→   111→ * - **Scope.** Designed for single-process local use (no cross-process
   112→   112→ *   locking); concurrent writers in separate processes are not supported.
   113→   113→ *
   114→   114→ * The constructor performs no filesystem access; directories are created
   115→   115→ * lazily on first publish, so read-only use against an empty/missing root
   116→   116→ * yields empty results rather than side effects.
   117→   117→ */
   118→   118→export class RegistryStore {
   119→   119→  /** Filesystem layout of this store (for CLI display and tooling). */
   120→   120→  public readonly paths: RegistryPaths;
   121→   121→
   122→   122→  constructor(rootDir: string) {
   123→   123→    if (typeof rootDir !== 'string' || rootDir.length === 0) {
   124→   124→      throw new TypeError('RegistryStore: rootDir must be a non-empty string');
   125→   125→    }
   126→   126→    const root = resolve(rootDir);
   127→   127→    this.paths = {
   128→   128→      root,
   129→   129→      objects: join(root, OBJECTS_DIR),
   130→   130→      packages: join(root, PACKAGES_DIR),
   131→   131→      index: join(root, INDEX_FILE),
   132→   132→    };
   133→   133→  }
   134→   134→
   135→   135→  // ------------------------------------------------------------------ publish
   136→   136→
   137→   137→  /**
   138→   138→   * Publish an immutable contract version.
   139→   139→   *
   140→   140→   * The version coordinates come from the package name's final segment
   141→   141→   * (`'payments.v1'` → version `v1`); names without one require
   142→   142→   * `opts.version`. Behavior when `packages/<base>/<version>/` already
   143→   143→   * exists:
   144→   144→   * - identical content → idempotent no-op, returns the originally stored
   145→   145→   *   meta (later publish metadata is ignored — versions are immutable);
   146→   146→   * - different content → {@link RegistryError} `'immutable'`.
   147→   147→   *
   148→   148→   * Every publish also rebuilds `index.json` from disk. `meta` fields
   149→   149→   * (`owner`, `description`, `repository`) and `opts.publishTime` are stored
   150→   150→   * verbatim; `publishedAt` is never generated implicitly.
   151→   151→   *
   152→   152→   * Throws `'invalid-name'` / `'invalid-version'` for bad coordinates,
   153→   153→   * `'hash-conflict'` if the content-addressed object slot holds different
   154→   154→   * content, `'immutable'` on version conflicts, `'io'` on filesystem
   155→   155→   * failures.
   156→   156→   */
   157→   157→  public publish(ir: IRPackage, meta: PublishMeta = {}, opts: PublishOptions = {}): ContractMeta {
   158→   158→    if (!isPlainObject(ir)) {
   159→   159→      throw new TypeError('publish(): ir must be an IRPackage object');
   160→   160→    }
   161→   161→    if (typeof ir.name !== 'string') {
   162→   162→      throw new TypeError("publish(): ir.name must be a string");
   163→   163→    }
   164→   164→    assertValidPackageName(ir.name);
   165→   165→    if (!Array.isArray(ir.imports) || !ir.imports.every((x) => typeof x === 'string')) {
   166→   166→      throw new TypeError("publish(): ir.imports must be an array of strings");
   167→   167→    }
   168→   168→    this.assertPublishMeta(meta);
   169→   169→
   170→   170→    const { base, version: derived } = splitPackageVersion(ir.name);
   171→   171→    const version = this.resolvePublishVersion(ir.name, derived, opts);
   172→   172→
   173→   173→    const hash = hashPackage(ir);
   174→   174→    assertSha256Hex(hash, `computed hash of '${ir.name}'`);
   175→   175→
   176→   176→    const verDir = join(this.paths.packages, base, version);
   177→   177→    const pointerPath = join(verDir, POINTER_FILE);
   178→   178→    const metaPath = join(verDir, META_FILE);
   179→   179→
   180→   180→    const rawPointer = parseJsonFile<unknown>(pointerPath);
   181→   181→    if (rawPointer !== null) {
   182→   182→      const pointer = assertPointerShape(rawPointer, pointerPath);
   183→   183→      if (pointer.hash !== hash) {
   184→   184→        throw new RegistryError(
   185→   185→          'immutable',
   186→   186→          `${base}@${version} is already published with different content ` +
   187→   187→            `(stored ${pointer.hash}, incoming ${hash}). Published versions are immutable — ` +
   188→   188→            `publish a new version instead.`,
   189→   189→        );
   190→   190→      }
   191→   191→      // Idempotent republish of identical content: a true no-op.
   192→   192→      const rawMeta = parseJsonFile<unknown>(metaPath);
   193→   193→      if (rawMeta === null) {
   194→   194→        throw new RegistryError(
   195→   195→          'corrupt',
   196→   196→          `${pointerPath} exists but ${metaPath} is missing; registry state is inconsistent`,
   197→   197→        );
   198→   198→      }
   199→   199→      const existingMeta = assertMetaShape(rawMeta, metaPath);
   200→   200→      this.ensureObject(ir, hash); // self-heal a missing object, detect tampering
   201→   201→      return existingMeta;
   202→   202→    }
   203→   203→    if (pathExists(metaPath)) {
   204→   204→      throw new RegistryError(
   205→   205→        'corrupt',
   206→   206→        `${metaPath} exists without ${POINTER_FILE}; registry state is inconsistent`,
   207→   207→      );
   208→   208→    }
   209→   209→
   210→   210→    const contractMeta = this.buildMeta(ir, base, version, hash, meta, opts);
   211→   211→    // Write order matters: the content-addressed object must exist before
   212→   212→    // anything references it. Each individual write is atomic.
   213→   213→    this.ensureObject(ir, hash);
   214→   214→    atomicWriteFile(pointerPath, canonicalJson({ hash, package: ir.name } satisfies PointerFile));
   215→   215→    atomicWriteFile(metaPath, canonicalJson(contractMeta));
   216→   216→    this.rebuildIndex();
   217→   217→    return contractMeta;
   218→   218→  }
   219→   219→
   220→   220→  // ------------------------------------------------------------------ pulling
   221→   221→
   222→   222→  /**
   223→   223→   * Fetch one published version.
   224→   224→   *
   225→   225→   * Returns the parsed IR and its metadata. The stored artifact is re-hashed
   226→   226→   * on every pull: a mismatch between content and the recorded hash throws
   227→   227→   * `'corrupt'` (tamper detection). Unknown coordinates throw `'not-found'`.
   228→   228→   */
   229→   229→  public pull(packageName: string, version: string): { ir: IRPackage; meta: ContractMeta } {
   230→   230→    const { base } = this.baseOf(packageName);
   231→   231→    const ver = normalizeVersion(version);
   232→   232→    const verDir = join(this.paths.packages, base, ver);
   233→   233→    const pointer = this.requirePointer(join(verDir, POINTER_FILE), base, ver);
   234→   234→    const ir = this.readVerifiedObject(pointer.hash, base, ver);
   235→   235→    const meta = this.requireMeta(join(verDir, META_FILE), base, ver);
   236→   236→    if (meta.hash !== pointer.hash) {
   237→   237→      throw new RegistryError(
   238→   238→        'corrupt',
   239→   239→        `Meta for ${base}@${ver} records hash ${meta.hash} but the pointer records ${pointer.hash}`,
   240→   240→      );
   241→   241→    }
   242→   242→    return { ir, meta };
   243→   243→  }
   244→   244→
   245→   245→  /**
   246→   246→   * Re-hash the stored artifact for `packageName@version` and compare it to
   247→   247→   * the recorded content address. Returns `{ ok: true, hash }` on success;
   248→   248→   * throws `'corrupt'` on tampering, `'not-found'` when unknown.
   249→   249→   */
   250→   250→  public verify(packageName: string, version: string): { ok: true; hash: string } {
   251→   251→    const { base } = this.baseOf(packageName);
   252→   252→    const ver = normalizeVersion(version);
   253→   253→    const verDir = join(this.paths.packages, base, ver);
   254→   254→    const pointer = this.requirePointer(join(verDir, POINTER_FILE), base, ver);
   255→   255→    this.readVerifiedObject(pointer.hash, base, ver);
   256→   256→    return { ok: true, hash: pointer.hash };
   257→   257→  }
   258→   258→
   259→   259→  /**
   260→   260→   * Metadata for one version. Reads only `meta.json` (no integrity check —
   261→   261→   * use {@link verify} for that). Unknown versions throw `'not-found'`.
   262→   262→   */
   263→   263→  public inspect(packageName: string, version: string): ContractMeta {
   264→   264→    const { base } = this.baseOf(packageName);
   265→   265→    return this.inspectAt(base, normalizeVersion(version));
   266→   266→  }
   267→   267→
   268→   268→  /**
   269→   269→   * Latest published version of a package (numeric ordering, `v2 < v10`),
   270→   270→   * by base name — both `'payments'` and any `'payments vX'` name address
   271→   271→   * the same package. Throws `'not-found'` when nothing is published.
   272→   272→   */
   273→   273→  public latest(packageName: string): ContractMeta {
   274→   274→    const { base } = this.baseOf(packageName);
   275→   275→    const versions = this.versionsByBase(base);
   276→   276→    const newest = versions.at(-1);
   277→   277→    if (newest === undefined) {
   278→   278→      throw new RegistryError('not-found', `No versions published for package '${packageName}'`);
   279→   279→    }
   280→   280→    return this.inspectAt(base, newest);
   281→   281→  }
   282→   282→
   283→   283→  /**
   284→   284→   * All published versions of a package (by base name), ordered
   285→   285→   * oldest → newest. Empty array for unknown packages.
   286→   286→   */
   287→   287→  public versions(packageName: string): string[] {
   288→   288→    const { base } = this.baseOf(packageName);
   289→   289→    return this.versionsByBase(base);
   290→   290→  }
   291→   291→
   292→   292→  /**
   293→   293→   * All packages with their latest meta, one entry per base, sorted by base
   294→   294→   * name. Reads the index (fast path); empty array for a fresh store.
   295→   295→   */
   296→   296→  public list(): ContractMeta[] {
   297→   297→    const latestByBase = new Map<string, ContractMeta>();
   298→   298→    for (const meta of this.readIndex()) {
   299→   299→      const prev = latestByBase.get(meta.base);
   300→   300→      if (prev === undefined || compareVersions(meta.version, prev.version) > 0) {
   301→   301→        latestByBase.set(meta.base, meta);
   302→   302→      }
   303→   303→    }
   304→   304→    return [...latestByBase.values()].sort(compareMeta);
   305→   305→  }
   306→   306→
   307→   307→  /**
   308→   308→   * Substring search (case-insensitive) over package name, base,
   309→   309→   * description and owner, using the index. Results follow the canonical
   310→   310→   * deterministic `(base, version)` order — identical queries return
   311→   311→   * identical arrays. An empty query matches everything.
   312→   312→   */
   313→   313→  public search(query: string): ContractMeta[] {
   314→   314→    if (typeof query !== 'string') throw new TypeError('search(): query must be a string');
   315→   315→    const q = query.toLowerCase();
   316→   316→    return this.readIndex().filter((meta) => {
   317→   317→      if (meta.packageName.toLowerCase().includes(q)) return true;
   318→   318→      if (meta.base.toLowerCase().includes(q)) return true;
   319→   319→      if (meta.owner !== undefined && meta.owner.toLowerCase().includes(q)) return true;
   320→   320→      if (meta.description !== undefined && meta.description.toLowerCase().includes(q)) return true;
   321→   321→      return false;
   322→   322→    });
   323→   323→  }
   324→   324→
   325→   325→  // ------------------------------------------------------------------- graph
   326→   326→
   327→   327→  /**
   328→   328→   * Direct dependents of a package: index contracts whose recorded imports
   329→   329→   * include `packageName` **or** its base. Deterministic `(base, version)`
   330→   330→   * order. A contract is never its own dependent.
   331→   331→   */
   332→   332→  public dependents(packageName: string): ContractMeta[] {
   333→   333→    assertValidPackageName(packageName);
   334→   334→    const base = splitPackageVersion(packageName).base;
   335→   335→    return this.readIndex().filter(
   336→   336→      (meta) =>
   337→   337→        meta.packageName !== packageName &&
   338→   338→        (meta.imports.includes(packageName) || meta.imports.includes(base)),
   339→   339→    );
   340→   340→  }
   341→   341→
   342→   342→  /**
   343→   343→   * Transitive dependency closure of a package, breadth-first over the
   344→   344→   * imports recorded in each dependency's stored meta.
   345→   345→   *
   346→   346→   * - Import references appear in the result as recorded (full names like
   347→   347→   *   `'payments.v1'`, or bare bases like `'payments'`), in BFS discovery
   348→   348→   *   order, deduplicated; the root package itself is excluded.
   349→   349→   * - Bare-base imports resolve to the latest published version of that
   350→   350→   *   base; unpublished dependencies are included but not traversed.
   351→   351→   * - Cycle-safe: visited state is tracked by name and storage key, so
   352→   352→   *   `a → b → a` terminates.
   353→   353→   *
   354→   354→   * `version` selects the root version; defaults to the latest. Unknown
   355→   355→   * roots throw `'not-found'`.
   356→   356→   */
   357→   357→  public dependencies(packageName: string, version?: string): string[] {
   358→   358→    const { base } = this.baseOf(packageName);
   359→   359→    let rootVersion: string;
   360→   360→    if (version !== undefined) {
   361→   361→      rootVersion = normalizeVersion(version);
   362→   362→    } else {
   363→   363→      const newest = this.versionsByBase(base).at(-1);
   364→   364→      if (newest === undefined) {
   365→   365→        throw new RegistryError(
   366→   366→          'not-found',
   367→   367→          `No versions published for package '${packageName}'`,
   368→   368→        );
   369→   369→      }
   370→   370→      rootVersion = newest;
   371→   371→    }
   372→   372→    const rootMeta = this.inspectAt(base, rootVersion);
   373→   373→
   374→   374→    const seenNames = new Set<string>([rootMeta.packageName, rootMeta.base]);
   375→   375→    const seenKeys = new Set<string>([storageKey(base, rootVersion)]);
   376→   376→    const result: string[] = [];
   377→   377→    const queue: string[] = [...rootMeta.imports];
   378→   378→
   379→   379→    while (queue.length > 0) {
   380→   380→      const name = queue.shift();
   381→   381→      if (name === undefined) break; // unreachable given the length guard
   382→   382→      if (seenNames.has(name)) continue;
   383→   383→      seenNames.add(name);
   384→   384→      result.push(name);
   385→   385→      const meta = this.findImportMeta(name);
   386→   386→      if (meta === null) continue; // unpublished dependency: report, don't traverse
   387→   387→      const key = storageKey(meta.base, meta.version);
   388→   388→      if (seenKeys.has(key)) continue; // same package reached via another alias
   389→   389→      seenKeys.add(key);
   390→   390→      for (const imp of meta.imports) {
   391→   391→        if (!seenNames.has(imp)) queue.push(imp);
   392→   392→      }
   393→   393→    }
   394→   394→    return result;
   395→   395→  }

   397→   397→  // ----------------------------------------------------------------- private
   398→   398→
   399→   399→  /** Validate a caller-supplied package name and reduce it to its base. */
   400→   400→  private baseOf(packageName: string): { base: string } {
   401→   401→    assertValidPackageName(packageName);
   402→   402→    return { base: splitPackageVersion(packageName).base };
   403→   403→  }
   404→   404→
   405→   405→  /** Validate publish metadata values (defensive against JS callers). */
   406→   406→  private assertPublishMeta(meta: PublishMeta): void {
   407→   407→    for (const field of ['owner', 'description', 'repository'] as const) {
   408→   408→      const value = meta[field];
   409→   409→      if (value !== undefined && typeof value !== 'string') {
   410→   410→        throw new TypeError(`publish(): meta.${field} must be a string when provided`);
   411→   411→      }
   412→   412→    }
   413→   413→  }
   414→   414→
   415→   415→  /**
   416→   416→   * Decide the storage version for a publish: the name-derived version when
   417→   417→   * present (then `opts.version` must match it), otherwise `opts.version`
   418→   418→   * is required and normalized.
   419→   419→   */
   420→   420→  private resolvePublishVersion(
   421→   421→    packageName: string,
   422→   422→    derived: string,
   423→   423→    opts: PublishOptions,
   424→   424→  ): string {
   425→   425→    if (derived !== '') {
   426→   426→      if (opts.version !== undefined) {
   427→   427→        const given = normalizeVersion(opts.version);
   428→   428→        if (given !== derived) {
   429→   429→          throw new RegistryError(
   430→   430→            'invalid-version',
   431→   431→            `opts.version ${JSON.stringify(given)} does not match the version ` +
   432→   432→              `${JSON.stringify(derived)} derived from package name '${packageName}'`,
   433→   433→          );
   434→   434→        }
   435→   435→      }
   436→   436→      return derived;
   437→   437→    }
   438→   438→    if (opts.version === undefined) {
   439→   439→      throw new RegistryError(
   440→   440→        'invalid-version',
   441→   441→        `Package name '${packageName}' carries no version segment; pass opts.version (e.g. 'v1')`,
   442→   442→      );
   443→   443→    }
   444→   444→    return normalizeVersion(opts.version);
   445→   445→  }
   446→   446→
   447→   447→  /** Build the stored meta for a fresh publish. */
   448→   448→  private buildMeta(
   449→   449→    ir: IRPackage,
   450→   450→    base: string,
   451→   451→    version: string,
   452→   452→    hash: string,
   453→   453→    meta: PublishMeta,
   454→   454→    opts: PublishOptions,
   455→   455→  ): ContractMeta {
   456→   456→    const out: ContractMeta = {
   457→   457→      packageName: ir.name,
   458→   458→      base,
   459→   459→      version,
   460→   460→      hash,
   461→   461→      shortHash: hash.slice(0, 12),
   462→   462→      // IR contract guarantees sorted+deduped imports; enforce it so meta is
   463→   463→      // deterministic even for hand-built fixtures.
   464→   464→      imports: [...new Set(ir.imports)].sort(),
   465→   465→    };
   466→   466→    if (meta.owner !== undefined) out.owner = meta.owner;
   467→   467→    if (meta.description !== undefined) out.description = meta.description;
   468→   468→    if (meta.repository !== undefined) out.repository = meta.repository;
   469→   469→    if (opts.publishTime !== undefined) out.publishedAt = opts.publishTime;
   470→   470→    return out;
   471→   471→  }
   472→   472→
   473→   473→  /**
   474→   474→   * Path of the content-addressed object for `hash`. The hash is validated
   475→   475→   * before it touches a path (traversal-safe even for hashes read from disk).
   476→   476→   */
   477→   477→  private objectPath(hash: string): string {
   478→   478→    assertSha256Hex(hash, 'object hash');
   479→   479→    return join(this.paths.objects, hash.slice(0, 2), `${hash}.json`);
   480→   480→  }
   481→   481→
   482→   482→  /**
   483→   483→   * Write the content-addressed object if absent. If the slot already holds
   484→   484→   * content, it must be byte-equivalent (canonically) to the incoming IR —
   485→   485→   * anything else is a SHA-256 collision or store tampering and throws
   486→   486→   * `'hash-conflict'` rather than silently overwriting.
   487→   487→   */
   488→   488→  private ensureObject(ir: IRPackage, hash: string): string {
   489→   489→    const objectPath = this.objectPath(hash);
   490→   490→    const existing = parseJsonFile<unknown>(objectPath);
   491→   491→    if (existing !== null) {
   492→   492→      if (canonicalJson(existing) !== canonicalJson(ir)) {
   493→   493→        throw new RegistryError(
   494→   494→          'hash-conflict',
   495→   495→          `Content-addressed object ${objectPath} already holds different content; ` +
   496→   496→            `refusing to overwrite (SHA-256 collision or store tampering).`,
   497→   497→        );
   498→   498→      }
   499→   499→      return objectPath;
   500→   500→    }
   501→   501→    atomicWriteFile(objectPath, canonicalJson(ir));
   502→   502→    return objectPath;
   503→   503→  }
   504→   504→
   505→   505→  /** Read an object and verify it hashes to the recorded address. */
   506→   506→  private readVerifiedObject(hash: string, base: string, version: string): IRPackage {
   507→   507→    const objectPath = this.objectPath(hash);
   508→   508→    const stored = parseJsonFile<unknown>(objectPath);
   509→   509→    if (stored === null) {
   510→   510→      throw new RegistryError(
   511→   511→        'corrupt',
   512→   512→        `Object ${objectPath} referenced by ${base}@${version} is missing`,
   513→   513→      );
   514→   514→    }
   515→   515→    const actual = hashPackage(stored as IRPackage);
   516→   516→    if (actual !== hash) {
   517→   517→      throw new RegistryError(
   518→   518→        'corrupt',
   519→   519→        `Integrity check failed for ${base}@${version}: recorded hash ${hash} but content ` +
   520→   520→          `hashes to ${actual} (artifact tampered or corrupted)`,
   521→   521→      );
   522→   522→    }
   523→   523→    return stored as IRPackage;
   524→   524→  }
   525→   525→
   526→   526→  /** Read a pointer or throw `'not-found'` when absent. */
   527→   527→  private requirePointer(pointerPath: string, base: string, version: string): PointerFile {
   528→   528→    const raw = parseJsonFile<unknown>(pointerPath);
   529→   529→    if (raw === null) {
   530→   530→      throw new RegistryError(
   531→   531→        'not-found',
   532→   532→        `No contract published at ${base}@${version} (missing ${pointerPath})`,
   533→   533→      );
   534→   534→    }
   535→   535→    return assertPointerShape(raw, pointerPath);
   536→   536→  }
   537→   537→
   538→   538→  /** Read a meta file in a context where the pointer is known to exist. */
   539→   539→  private requireMeta(metaPath: string, base: string, version: string): ContractMeta {
   540→   540→    const raw = parseJsonFile<unknown>(metaPath);
   541→   541→    if (raw === null) {
   542→   542→      throw new RegistryError(
   543→   543→        'corrupt',
   544→   544→        `Meta file ${metaPath} missing for ${base}@${version}; registry state is inconsistent`,
   545→   545→      );
   546→   546→    }
   547→   547→    return assertMetaShape(raw, metaPath);
   548→   548→  }
   549→   549→
   550→   550→  /** Meta at `packages/<base>/<version>/meta.json`, with honest errors. */
   551→   551→  private inspectAt(base: string, version: string): ContractMeta {
   552→   552→    const verDir = join(this.paths.packages, base, version);
   553→   553→    const metaPath = join(verDir, META_FILE);
   554→   554→    const raw = parseJsonFile<unknown>(metaPath);
   555→   555→    if (raw !== null) return assertMetaShape(raw, metaPath);
   556→   556→    if (pathExists(join(verDir, POINTER_FILE))) {
   557→   557→      throw new RegistryError(
   558→   558→        'corrupt',
   559→   559→        `${verDir} has a pointer but no ${META_FILE}; registry state is inconsistent`,
   560→   560→      );
   561→   561→    }
   562→   562→    throw new RegistryError('not-found', `No contract published at ${base}@${version}`);
   563→   563→  }
   564→   564→
   565→   565→  /** Meta or `null` — used when absence is an expected, non-fatal outcome. */
   566→   566→  private inspectOrNull(base: string, version: string): ContractMeta | null {
   567→   567→    const metaPath = join(this.paths.packages, base, version, META_FILE);
   568→   568→    const raw = parseJsonFile<unknown>(metaPath);
   569→   569→    if (raw === null) return null;
   570→   570→    return assertMetaShape(raw, metaPath);
   571→   571→  }
   572→   572→
   573→   573→  /** Published versions under one base, oldest → newest. */
   574→   574→  private versionsByBase(base: string): string[] {
   575→   575→    assertValidPackageName(base); // defense in depth for stored-derived bases
   576→   576→    const baseDir = join(this.paths.packages, base);
   577→   577→    const dirents = readdirDirentsOrNull(baseDir);
   578→   578→    if (dirents === null) return [];
   579→   579→    const out: string[] = [];
   580→   580→    for (const dirent of dirents) {
   581→   581→      if (!dirent.isDirectory()) continue;
   582→   582→      if (!VERSION_DIR_RE.test(dirent.name)) continue;
   583→   583→      if (pathExists(join(baseDir, dirent.name, POINTER_FILE))) out.push(dirent.name);
   584→   584→    }
   585→   585→    out.sort(compareVersions);
   586→   586→    return out;
   587→   587→  }
   588→   588→
   589→   589→  /** Resolve an import reference (full name or bare base) to a stored meta. */
   590→   590→  private findImportMeta(name: string): ContractMeta | null {
   591→   591→    if (!isValidPackageName(name)) {
   592→   592→      throw new RegistryError(
   593→   593→        'corrupt',
   594→   594→        `Stored import reference ${JSON.stringify(name)} is not a valid package name`,
   595→   595→      );
   596→   596→    }
   597→   597→    const { base, version } = splitPackageVersion(name);
   598→   598→    if (version !== '') return this.inspectOrNull(base, version);
   599→   599→    // Bare-base import: resolve to the latest published version.
   600→   600→    const newest = this.versionsByBase(base).at(-1);
   601→   601→    return newest === undefined ? null : this.inspectOrNull(base, newest);
   602→   602→  }
   603→   603→
   604→   604→  /** All metas from `index.json`, re-sorted into canonical order. */
   605→   605→  private readIndex(): ContractMeta[] {
   606→   606→    const raw = parseJsonFile<unknown>(this.paths.index);
   607→   607→    if (raw === null) return [];
   608→   608→    if (!isPlainObject(raw) || !Array.isArray(raw.contracts)) {
   609→   609→      throw new RegistryError(
   610→   610→        'corrupt',
   611→   611→        `${this.paths.index}: expected {"contracts": ContractMeta[]}`,
   612→   612→      );
   613→   613→    }
   614→   614→    return raw.contracts
   615→   615→      .map((entry, i) => assertMetaShape(entry, `${this.paths.index}[contracts][${i}]`))
   616→   616→      .sort(compareMeta);
   617→   617→  }
   618→   618→
   619→   619→  /** Rebuild `index.json` from disk (atomic). */
   620→   620→  private rebuildIndex(): void {
   621→   621→    const contracts: IndexFile = { contracts: this.scanMetas() };
   622→   622→    atomicWriteFile(this.paths.index, canonicalJson(contracts));
   623→   623→  }
   624→   624→
   625→   625→  /**
   626→   626→   * Scan every `packages/<base>/<version>/meta.json` into a sorted array.
   627→   627→   * Lenient by design: corrupt or partial entries are skipped so one bad
   628→   628→   * file cannot block publishing; direct accessors (pull/inspect/verify)
   629→   629→   * still surface those errors loudly.
   630→   630→   */
   631→   631→  private scanMetas(): ContractMeta[] {
   632→   632→    const dirents = readdirDirentsOrNull(this.paths.packages);
   633→   633→    if (dirents === null) return [];
   634→   634→    const out: ContractMeta[] = [];
   635→   635→    for (const baseDir of dirents) {
   636→   636→      if (!baseDir.isDirectory()) continue;
   637→   637→      const versions = readdirDirentsOrNull(join(this.paths.packages, baseDir.name));
   638→   638→      if (versions === null) continue;
   639→   639→      for (const verDir of versions) {
   640→   640→        if (!verDir.isDirectory()) continue;
   641→   641→        const metaPath = join(this.paths.packages, baseDir.name, verDir.name, META_FILE);
   642→   642→        try {
   643→   643→          const raw = parseJsonFile<unknown>(metaPath);
   644→   644→          if (raw === null) continue;
   645→   645→          out.push(assertMetaShape(raw, metaPath));
   646→   646→        } catch {
   647→   647→          continue; // corrupt unrelated entry — skip, don't block the index
   648→   648→        }
   649→   649→      }
   650→   650→    }
   651→   651→    return out.sort(compareMeta);
   652→   652→  }
