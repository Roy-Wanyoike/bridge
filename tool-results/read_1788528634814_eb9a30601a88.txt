    16→/** Pointer file inside `packages/<base>/<version>/`. */
    17→const POINTER_FILE = 'contract.json';
    18→/** Metadata file inside `packages/<base>/<version>/`. */
    19→const META_FILE = 'meta.json';
    20→/** Store-wide index file. */
    21→const INDEX_FILE = 'index.json';
    22→/** Object store directory name. */
    23→const OBJECTS_DIR = 'objects';
    24→/** Package tree directory name. */
    25→const PACKAGES_DIR = 'packages';
    26→
    27→/** Version directory names are always canonical `v<digits>`. */
    28→const VERSION_DIR_RE = /^v\d+$/;
    29→
    30→/** Shape of `packages/<base>/<version>/contract.json`. */
    31→interface PointerFile {
    32→  hash: string;
    33→  package: string;
    34→}
    35→
    36→/** Shape of `index.json`. */
    37→interface IndexFile {
    38→  contracts: ContractMeta[];
    39→}
    40→
    41→function isPlainObject(v: unknown): v is Record<string, unknown> {
    42→  return typeof v === 'object' && v !== null && !Array.isArray(v);
    43→}
    44→
    45→function storageKey(base: string, version: string): string {
    46→  return `${base}@${version}`;
    47→}
    48→
    49→/** Canonical deterministic order for contract metas: (base, version). */
    50→function compareMeta(a: ContractMeta, b: ContractMeta): number {
    51→  if (a.base !== b.base) return a.base < b.base ? -1 : 1;
    52→  return compareVersions(a.version, b.version);
    53→}
    54→
    55→/** Validate an untrusted parsed meta record; `'corrupt'` on any deviation. */
    56→function assertMetaShape(value: unknown, file: string): ContractMeta {
    57→  if (!isPlainObject(value)) {
    58→    throw new RegistryError('corrupt', `${file}: expected a JSON object`);
    59→  }
    60→  for (const field of ['packageName', 'base', 'version', 'hash', 'shortHash'] as const) {
    61→    if (typeof value[field] !== 'string') {
    62→      throw new RegistryError('corrupt', `${file}: field '${field}' must be a string`);
    63→    }
    64→  }
    65→  if (!Array.isArray(value.imports) || !value.imports.every((x) => typeof x === 'string')) {
    66→    throw new RegistryError('corrupt', `${file}: field 'imports' must be an array of strings`);
    67→  }
    68→  assertSha256Hex(value.hash, `${file}: meta.hash`);
    69→  return value as unknown as ContractMeta;
    70→}
    71→
    72→/** Validate an untrusted parsed pointer record; `'corrupt'` on any deviation. */
    73→function assertPointerShape(value: unknown, file: string): PointerFile {
    74→  if (!isPlainObject(value)) {
    75→    throw new RegistryError('corrupt', `${file}: expected a JSON object`);
    76→  }
    77→  const hash = assertSha256Hex(value.hash, `${file}: pointer.hash`);
    78→  if (typeof value.package !== 'string' || !isValidPackageName(value.package)) {
    79→    throw new RegistryError(
    80→      'corrupt',
    81→      `${file}: pointer.package ${JSON.stringify(value.package)} is not a valid package name`,
    82→    );
    83→  }
    84→  return { hash, package: value.package };
    85→}

    87→/**
    88→ * Local, content-addressed registry of immutable contract versions.
    89→ *
    90→ * Storage layout (all writes atomic: hidden same-directory tmp file +
    91→ * rename; all JSON deterministic — sorted keys via `canonicalJson`):
    92→ *
    93→ * ```
    94→ * <root>/
    95→ *   objects/<hash[0:2]>/<hash>.json          ← canonicalJson(ir), content-addressed
    96→ *   packages/<base>/<version>/contract.json  ← { "hash": ..., "package": <name> }
    97→ *   packages/<base>/<version>/meta.json      ← ContractMeta
    98→ *   index.json                               ← { "contracts": [ContractMeta...] }
    99→ * ```
   100→ *
   101→ * Key semantics:
   102→ * - **Content addressing.** A version's identity is `hashPackage(ir)`.
   103→ *   Republishing identical content is a no-op returning the original meta;
   104→ *   republishing different content under an existing base/version throws
   105→ *   `'immutable'` — published versions are never mutated.
   106→ * - **Lookups are keyed by `(base, version)`.** The base is the package name
   107→ *   minus a version-shaped final segment (`'payments.v1'` and `'payments'`
   108→ *   both address the base `'payments'`).
   109→ * - **Path safety.** Every name/version/hash that reaches a filesystem path
   110→ *   is validated first, whether it came from a caller or from stored data.
   111→ * - **Scope.** Designed for single-process local use (no cross-process
   112→ *   locking); concurrent writers in separate processes are not supported.
   113→ *
   114→ * The constructor performs no filesystem access; directories are created
   115→ * lazily on first publish, so read-only use against an empty/missing root
   116→ * yields empty results rather than side effects.
   117→ */
   118→export class RegistryStore {
   119→  /** Filesystem layout of this store (for CLI display and tooling). */
   120→  public readonly paths: RegistryPaths;
   121→
   122→  constructor(rootDir: string) {
   123→    if (typeof rootDir !== 'string' || rootDir.length === 0) {
   124→      throw new TypeError('RegistryStore: rootDir must be a non-empty string');
   125→    }
   126→    const root = resolve(rootDir);
   127→    this.paths = {
   128→      root,
   129→      objects: join(root, OBJECTS_DIR),
   130→      packages: join(root, PACKAGES_DIR),
   131→      index: join(root, INDEX_FILE),
   132→    };
   133→  }
   134→
   135→  // ------------------------------------------------------------------ publish
   136→
   137→  /**
   138→   * Publish an immutable contract version.
   139→   *
   140→   * The version coordinates come from the package name's final segment
   141→   * (`'payments.v1'` → version `v1`); names without one require
   142→   * `opts.version`. Behavior when `packages/<base>/<version>/` already
   143→   * exists:
   144→   * - identical content → idempotent no-op, returns the originally stored
   145→   *   meta (later publish metadata is ignored — versions are immutable);
   146→   * - different content → {@link RegistryError} `'immutable'`.
   147→   *
   148→   * Every publish also rebuilds `index.json` from disk. `meta` fields
   149→   * (`owner`, `description`, `repository`) and `opts.publishTime` are stored
   150→   * verbatim; `publishedAt` is never generated implicitly.
   151→   *
   152→   * Throws `'invalid-name'` / `'invalid-version'` for bad coordinates,
   153→   * `'hash-conflict'` if the content-addressed object slot holds different
   154→   * content, `'immutable'` on version conflicts, `'io'` on filesystem
   155→   * failures.
   156→   */
   157→  public publish(ir: IRPackage, meta: PublishMeta = {}, opts: PublishOptions = {}): ContractMeta {
   158→    if (!isPlainObject(ir)) {
   159→      throw new TypeError('publish(): ir must be an IRPackage object');
   160→    }
   161→    if (typeof ir.name !== 'string') {
   162→      throw new TypeError("publish(): ir.name must be a string");
   163→    }
   164→    assertValidPackageName(ir.name);
   165→    if (!Array.isArray(ir.imports) || !ir.imports.every((x) => typeof x === 'string')) {
   166→      throw new TypeError("publish(): ir.imports must be an array of strings");
   167→    }
   168→    this.assertPublishMeta(meta);
   169→
   170→    const { base, version: derived } = splitPackageVersion(ir.name);
   171→    const version = this.resolvePublishVersion(ir.name, derived, opts);
   172→
   173→    const hash = hashPackage(ir);
   174→    assertSha256Hex(hash, `computed hash of '${ir.name}'`);
   175→
   176→    const verDir = join(this.paths.packages, base, version);
   177→    const pointerPath = join(verDir, POINTER_FILE);
   178→    const metaPath = join(verDir, META_FILE);
   179→
   180→    const rawPointer = parseJsonFile<unknown>(pointerPath);
   181→    if (rawPointer !== null) {
   182→      const pointer = assertPointerShape(rawPointer, pointerPath);
   183→      if (pointer.hash !== hash) {
   184→        throw new RegistryError(
   185→          'immutable',
   186→          `${base}@${version} is already published with different content ` +
   187→            `(stored ${pointer.hash}, incoming ${hash}). Published versions are immutable — ` +
   188→            `publish a new version instead.`,
   189→        );
   190→      }
   191→      // Idempotent republish of identical content: a true no-op.
   192→      const rawMeta = parseJsonFile<unknown>(metaPath);
   193→      if (rawMeta === null) {
   194→        throw new RegistryError(
   195→          'corrupt',
   196→          `${pointerPath} exists but ${metaPath} is missing; registry state is inconsistent`,
   197→        );
   198→      }
   199→      const existingMeta = assertMetaShape(rawMeta, metaPath);
   200→      this.ensureObject(ir, hash); // self-heal a missing object, detect tampering
   201→      return existingMeta;
   202→    }
   203→    if (pathExists(metaPath)) {
   204→      throw new RegistryError(
   205→        'corrupt',
   206→        `${metaPath} exists without ${POINTER_FILE}; registry state is inconsistent`,
   207→      );
   208→    }
   209→
   210→    const contractMeta = this.buildMeta(ir, base, version, hash, meta, opts);
   211→    // Write order matters: the content-addressed object must exist before
   212→    // anything references it. Each individual write is atomic.
   213→    this.ensureObject(ir, hash);
   214→    atomicWriteFile(pointerPath, canonicalJson({ hash, package: ir.name } satisfies PointerFile));
   215→    atomicWriteFile(metaPath, canonicalJson(contractMeta));
   216→    this.rebuildIndex();
   217→    return contractMeta;
   218→  }
   219→
   220→  // ------------------------------------------------------------------ pulling
   221→
   222→  /**
   223→   * Fetch one published version.
   224→   *
   225→   * Returns the parsed IR and its metadata. The stored artifact is re-hashed
   226→   * on every pull: a mismatch between content and the recorded hash throws
   227→   * `'corrupt'` (tamper detection). Unknown coordinates throw `'not-found'`.
   228→   */
   229→  public pull(packageName: string, version: string): { ir: IRPackage; meta: ContractMeta } {
   230→    const { base } = this.baseOf(packageName);
   231→    const ver = normalizeVersion(version);
   232→    const verDir = join(this.paths.packages, base, ver);
   233→    const pointer = this.requirePointer(join(verDir, POINTER_FILE), base, ver);
   234→    const ir = this.readVerifiedObject(pointer.hash, base, ver);
   235→    const meta = this.requireMeta(join(verDir, META_FILE), base, ver);
   236→    if (meta.hash !== pointer.hash) {
   237→      throw new RegistryError(
   238→        'corrupt',
   239→        `Meta for ${base}@${ver} records hash ${meta.hash} but the pointer records ${pointer.hash}`,
   240→      );
   241→    }
   242→    return { ir, meta };
   243→  }
   244→
   245→  /**
   246→   * Re-hash the stored artifact for `packageName@version` and compare it to
   247→   * the recorded content address. Returns `{ ok: true, hash }` on success;
   248→   * throws `'corrupt'` on tampering, `'not-found'` when unknown.
   249→   */
   250→  public verify(packageName: string, version: string): { ok: true; hash: string } {
   251→    const { base } = this.baseOf(packageName);
   252→    const ver = normalizeVersion(version);
   253→    const verDir = join(this.paths.packages, base, ver);
   254→    const pointer = this.requirePointer(join(verDir, POINTER_FILE), base, ver);
   255→    this.readVerifiedObject(pointer.hash, base, ver);
   256→    return { ok: true, hash: pointer.hash };
   257→  }
   258→
   259→  /**
   260→   * Metadata for one version. Reads only `meta.json` (no integrity check —
   261→   * use {@link verify} for that). Unknown versions throw `'not-found'`.
   262→   */
   263→  public inspect(packageName: string, version: string): ContractMeta {
   264→    const { base } = this.baseOf(packageName);
   265→    return this.inspectAt(base, normalizeVersion(version));
   266→  }
   267→
   268→  /**
   269→   * Latest published version of a package (numeric ordering, `v2 < v10`),
   270→   * by base name — both `'payments'` and any `'payments vX'` name address
   271→   * the same package. Throws `'not-found'` when nothing is published.
   272→   */
   273→  public latest(packageName: string): ContractMeta {
   274→    const { base } = this.baseOf(packageName);
   275→    const versions = this.versionsByBase(base);
   276→    const newest = versions.at(-1);
   277→    if (newest === undefined) {
   278→      throw new RegistryError('not-found', `No versions published for package '${packageName}'`);
   279→    }
   280→    return this.inspectAt(base, newest);
   281→  }
   282→
   283→  /**
   284→   * All published versions of a package (by base name), ordered
   285→   * oldest → newest. Empty array for unknown packages.
   286→   */
   287→  public versions(packageName: string): string[] {
   288→    const { base } = this.baseOf(packageName);
   289→    return this.versionsByBase(base);
   290→  }
   291→
   292→  /**
   293→   * All packages with their latest meta, one entry per base, sorted by base
   294→   * name. Reads the index (fast path); empty array for a fresh store.
   295→   */
   296→  public list(): ContractMeta[] {
   297→    const latestByBase = new Map<string, ContractMeta>();
   298→    for (const meta of this.readIndex()) {
   299→      const prev = latestByBase.get(meta.base);
   300→      if (prev === undefined || compareVersions(meta.version, prev.version) > 0) {
   301→        latestByBase.set(meta.base, meta);
   302→      }
   303→    }
   304→    return [...latestByBase.values()].sort(compareMeta);
   305→  }
   306→
   307→  /**
   308→   * Substring search (case-insensitive) over package name, base,
   309→   * description and owner, using the index. Results follow the canonical
   310→   * deterministic `(base, version)` order — identical queries return
   311→   * identical arrays. An empty query matches everything.
   312→   */
   313→  public search(query: string): ContractMeta[] {
   314→    if (typeof query !== 'string') throw new TypeError('search(): query must be a string');
   315→    const q = query.toLowerCase();
   316→    return this.readIndex().filter((meta) => {
   317→      if (meta.packageName.toLowerCase().includes(q)) return true;
   318→      if (meta.base.toLowerCase().includes(q)) return true;
   319→      if (meta.owner !== undefined && meta.owner.toLowerCase().includes(q)) return true;
   320→      if (meta.description !== undefined && meta.description.toLowerCase().includes(q)) return true;
   321→      return false;
   322→    });
   323→  }

   325→  // ------------------------------------------------------------------- graph
   326→
   327→  /**
   328→   * Direct dependents of a package: index contracts whose recorded imports
   329→   * include `packageName` **or** its base. Deterministic `(base, version)`
   330→   * order. A contract is never its own dependent.
   331→   */
   332→  public dependents(packageName: string): ContractMeta[] {
   333→    assertValidPackageName(packageName);
   334→    const base = splitPackageVersion(packageName).base;
   335→    return this.readIndex().filter(
   336→      (meta) =>
   337→        meta.packageName !== packageName &&
   338→        (meta.imports.includes(packageName) || meta.imports.includes(base)),
   339→    );
   340→  }
   341→
   342→  /**
   343→   * Transitive dependency closure of a package, breadth-first over the
   344→   * imports recorded in each dependency's stored meta.
   345→   *
   346→   * - Import references appear in the result as recorded (full names like
   347→   *   `'payments.v1'`, or bare bases like `'payments'`), in BFS discovery
   348→   *   order, deduplicated; the root package itself is excluded.
   349→   * - Bare-base imports resolve to the latest published version of that
   350→   *   base; unpublished dependencies are included but not traversed.
   351→   * - Cycle-safe: visited state is tracked by name and storage key, so
   352→   *   `a → b → a` terminates.
   353→   *
   354→   * `version` selects the root version; defaults to the latest. Unknown
   355→   * roots throw `'not-found'`.
   356→   */
   357→  public dependencies(packageName: string, version?: string): string[] {
   358→    const { base } = this.baseOf(packageName);
   359→    let rootVersion: string;
   360→    if (version !== undefined) {
   361→      rootVersion = normalizeVersion(version);
   362→    } else {
   363→      const newest = this.versionsByBase(base).at(-1);
   364→      if (newest === undefined) {
   365→        throw new RegistryError(
   366→          'not-found',
   367→          `No versions published for package '${packageName}'`,
   368→        );
   369→      }
   370→      rootVersion = newest;
   371→    }
   372→    const rootMeta = this.inspectAt(base, rootVersion);
   373→
   374→    const seenNames = new Set<string>([rootMeta.packageName, rootMeta.base]);
   375→    const seenKeys = new Set<string>([storageKey(base, rootVersion)]);
   376→    const result: string[] = [];
   377→    const queue: string[] = [...rootMeta.imports];
   378→
   379→    while (queue.length > 0) {
   380→      const name = queue.shift();
   381→      if (name === undefined) break; // unreachable given the length guard
   382→      if (seenNames.has(name)) continue;
   383→      seenNames.add(name);
   384→      result.push(name);
   385→      const meta = this.findImportMeta(name);
   386→      if (meta === null) continue; // unpublished dependency: report, don't traverse
   387→      const key = storageKey(meta.base, meta.version);
   388→      if (seenKeys.has(key)) continue; // same package reached via another alias
   389→      seenKeys.add(key);
   390→      for (const imp of meta.imports) {
   391→        if (!seenNames.has(imp)) queue.push(imp);
   392→      }
   393→    }
   394→    return result;
   395→  }

   399→  /** Validate a caller-supplied package name and reduce it to its base. */
   400→  private baseOf(packageName: string): { base: string } {
   401→    assertValidPackageName(packageName);
   402→    return { base: splitPackageVersion(packageName).base };

   405→  /** Validate publish metadata values (defensive against JS callers). */
   406→  private assertPublishMeta(meta: PublishMeta): void {
   407→    for (const field of ['owner', 'description', 'repository'] as const) {
   408→      const value = meta[field];
   409→      if (value !== undefined && typeof value !== 'string') {
   410→        throw new TypeError(`publish(): meta.${field} must be a string when provided`);
   411→      }
   412→    }
   413→  }

   415→  /**
   416→   * Decide the storage version for a publish: the name-derived version when
   417→   * present (then `opts.version` must match it), otherwise `opts.version`
   418→   * is required and normalized.
   419→   */
   420→  private resolvePublishVersion(
   421→    packageName: string,
   422→    derived: string,
   423→    opts: PublishOptions,
   424→  ): string {
   425→    if (derived !== '') {
   426→      if (opts.version !== undefined) {
   427→        const given = normalizeVersion(opts.version);
   428→        if (given !== derived) {
   429→          throw new RegistryError(
   430→            'invalid-version',
   431→            `opts.version ${JSON.stringify(given)} does not match the version ` +
   432→              `${JSON.stringify(derived)} derived from package name '${packageName}'`,
   433→          );
   434→        }
   435→      }
   436→      return derived;
   437→    }
   438→    if (opts.version === undefined) {
   439→      throw new RegistryError(
   440→        'invalid-version',
   441→        `Package name '${packageName}' carries no version segment; pass opts.version (e.g. 'v1')`,
   442→      );
   443→    }
   444→    return normalizeVersion(opts.version);
   445→  }
   446→
   447→  /** Build the stored meta for a fresh publish. */
   448→  private buildMeta(
   449→    ir: IRPackage,
   450→    base: string,
   451→    version: string,
   452→    hash: string,
   453→    meta: PublishMeta,
   454→    opts: PublishOptions,
   455→  ): ContractMeta {
   456→    const out: ContractMeta = {
   457→      packageName: ir.name,
   458→      base,
   459→      version,
   460→      hash,
   461→      shortHash: hash.slice(0, 12),
   462→      // IR contract guarantees sorted+deduped imports; enforce it so meta is
   463→      // deterministic even for hand-built fixtures.
   464→      imports: [...new Set(ir.imports)].sort(),
   465→    };
   466→    if (meta.owner !== undefined) out.owner = meta.owner;
   467→    if (meta.description !== undefined) out.description = meta.description;
   468→    if (meta.repository !== undefined) out.repository = meta.repository;
   469→    if (opts.publishTime !== undefined) out.publishedAt = opts.publishTime;
   470→    return out;
   471→  }
   472→
   473→  /**
   474→   * Path of the content-addressed object for `hash`. The hash is validated
   475→   * before it touches a path (traversal-safe even for hashes read from disk).
   476→   */
   477→  private objectPath(hash: string): string {
   478→    assertSha256Hex(hash, 'object hash');
   479→    return join(this.paths.objects, hash.slice(0, 2), `${hash}.json`);
   480→  }
   481→
   482→  /**
   483→   * Write the content-addressed object if absent. If the slot already holds
   484→   * content, it must be byte-equivalent (canonically) to the incoming IR —
   485→   * anything else is a SHA-256 collision or store tampering and throws
   486→   * `'hash-conflict'` rather than silently overwriting.
   487→   */
   488→  private ensureObject(ir: IRPackage, hash: string): string {
   489→    const objectPath = this.objectPath(hash);
   490→    const existing = parseJsonFile<unknown>(objectPath);
   491→    if (existing !== null) {
   492→      if (canonicalJson(existing) !== canonicalJson(ir)) {
   493→        throw new RegistryError(
   494→          'hash-conflict',
   495→          `Content-addressed object ${objectPath} already holds different content; ` +
   496→            `refusing to overwrite (SHA-256 collision or store tampering).`,
   497→        );
   498→      }
   499→      return objectPath;
   500→    }
   501→    atomicWriteFile(objectPath, canonicalJson(ir));
   502→    return objectPath;
   503→  }
   504→
   505→  /** Read an object and verify it hashes to the recorded address. */
   506→  private readVerifiedObject(hash: string, base: string, version: string): IRPackage {
   507→    const objectPath = this.objectPath(hash);
   508→    const stored = parseJsonFile<unknown>(objectPath);
   509→    if (stored === null) {
   510→      throw new RegistryError(
   511→        'corrupt',
   512→        `Object ${objectPath} referenced by ${base}@${version} is missing`,
   513→      );
   514→    }
   515→    const actual = hashPackage(stored as IRPackage);
   516→    if (actual !== hash) {
   517→      throw new RegistryError(
   518→        'corrupt',
   519→        `Integrity check failed for ${base}@${version}: recorded hash ${hash} but content ` +
   520→          `hashes to ${actual} (artifact tampered or corrupted)`,
   521→      );
   522→    }
   523→    return stored as IRPackage;
   524→  }
   525→
   526→  /** Read a pointer or throw `'not-found'` when absent. */
   527→  private requirePointer(pointerPath: string, base: string, version: string): PointerFile {
   528→    const raw = parseJsonFile<unknown>(pointerPath);
   529→    if (raw === null) {
   530→      throw new RegistryError(
   531→        'not-found',
   532→        `No contract published at ${base}@${version} (missing ${pointerPath})`,
   533→      );
   534→    }
   535→    return assertPointerShape(raw, pointerPath);
   536→  }
   537→
   538→  /** Read a meta file in a context where the pointer is known to exist. */
   539→  private requireMeta(metaPath: string, base: string, version: string): ContractMeta {
   540→    const raw = parseJsonFile<unknown>(metaPath);
   541→    if (raw === null) {
   542→      throw new RegistryError(
   543→        'corrupt',
   544→        `Meta file ${metaPath} missing for ${base}@${version}; registry state is inconsistent`,
   545→      );
   546→    }
   547→    return assertMetaShape(raw, metaPath);
   548→  }
   549→
   550→  /** Meta at `packages/<base>/<version>/meta.json`, with honest errors. */
   551→  private inspectAt(base: string, version: string): ContractMeta {
   552→    const verDir = join(this.paths.packages, base, version);
   553→    const metaPath = join(verDir, META_FILE);
   554→    const raw = parseJsonFile<unknown>(metaPath);
   555→    if (raw !== null) return assertMetaShape(raw, metaPath);
   556→    if (pathExists(join(verDir, POINTER_FILE))) {
   557→      throw new RegistryError(
   558→        'corrupt',
   559→        `${verDir} has a pointer but no ${META_FILE}; registry state is inconsistent`,
   560→      );
   561→    }
   562→    throw new RegistryError('not-found', `No contract published at ${base}@${version}`);
   563→  }
   564→
   565→  /** Meta or `null` — used when absence is an expected, non-fatal outcome. */
   566→  private inspectOrNull(base: string, version: string): ContractMeta | null {
   567→    const metaPath = join(this.paths.packages, base, version, META_FILE);
   568→    const raw = parseJsonFile<unknown>(metaPath);
   569→    if (raw === null) return null;
   570→    return assertMetaShape(raw, metaPath);
   571→  }
   572→
   573→  /** Published versions under one base, oldest → newest. */
   574→  private versionsByBase(base: string): string[] {
   575→    assertValidPackageName(base); // defense in depth for stored-derived bases
   576→    const baseDir = join(this.paths.packages, base);
   577→    const dirents = readdirDirentsOrNull(baseDir);
   578→    if (dirents === null) return [];
   579→    const out: string[] = [];
   580→    for (const dirent of dirents) {
   581→      if (!dirent.isDirectory()) continue;
   582→      if (!VERSION_DIR_RE.test(dirent.name)) continue;
   583→      if (pathExists(join(baseDir, dirent.name, POINTER_FILE))) out.push(dirent.name);
   584→    }
   585→    out.sort(compareVersions);
   586→    return out;
   587→  }
   588→
   589→  /** Resolve an import reference (full name or bare base) to a stored meta. */
   590→  private findImportMeta(name: string): ContractMeta | null {
   591→    if (!isValidPackageName(name)) {
   592→      throw new RegistryError(
   593→        'corrupt',
   594→        `Stored import reference ${JSON.stringify(name)} is not a valid package name`,
   595→      );
   596→    }
   597→    const { base, version } = splitPackageVersion(name);
   598→    if (version !== '') return this.inspectOrNull(base, version);
   599→    // Bare-base import: resolve to the latest published version.
   600→    const newest = this.versionsByBase(base).at(-1);
   601→    return newest === undefined ? null : this.inspectOrNull(base, newest);
   602→  }
   603→
   604→  /** All metas from `index.json`, re-sorted into canonical order. */
   605→  private readIndex(): ContractMeta[] {
   606→    const raw = parseJsonFile<unknown>(this.paths.index);
   607→    if (raw === null) return [];
   608→    if (!isPlainObject(raw) || !Array.isArray(raw.contracts)) {
   609→      throw new RegistryError(
   610→        'corrupt',
   611→        `${this.paths.index}: expected {"contracts": ContractMeta[]}`,
   612→      );
   613→    }
   614→    return raw.contracts
   615→      .map((entry, i) => assertMetaShape(entry, `${this.paths.index}[contracts][${i}]`))
   616→      .sort(compareMeta);
   617→  }
   618→
   619→  /** Rebuild `index.json` from disk (atomic). */
   620→  private rebuildIndex(): void {
   621→    const contracts: IndexFile = { contracts: this.scanMetas() };
   622→    atomicWriteFile(this.paths.index, canonicalJson(contracts));
   623→  }
   624→
   625→  /**
   626→   * Scan every `packages/<base>/<version>/meta.json` into a sorted array.
   627→   * Lenient by design: corrupt or partial entries are skipped so one bad
   628→   * file cannot block publishing; direct accessors (pull/inspect/verify)
   629→   * still surface those errors loudly.
   630→   */
   631→  private scanMetas(): ContractMeta[] {
   632→    const dirents = readdirDirentsOrNull(this.paths.packages);
   633→    if (dirents === null) return [];
   634→    const out: ContractMeta[] = [];
   635→    for (const baseDir of dirents) {
   636→      if (!baseDir.isDirectory()) continue;
   637→      const versions = readdirDirentsOrNull(join(this.paths.packages, baseDir.name));
   638→      if (versions === null) continue;
   639→      for (const verDir of versions) {
   640→        if (!verDir.isDirectory()) continue;
   641→        const metaPath = join(this.paths.packages, baseDir.name, verDir.name, META_FILE);
   642→        try {
   643→          const raw = parseJsonFile<unknown>(metaPath);
   644→          if (raw === null) continue;
   645→          out.push(assertMetaShape(raw, metaPath));
   646→        } catch {
   647→          continue; // corrupt unrelated entry — skip, don't block the index
   648→        }
   649→      }
   650→    }
   651→    return out.sort(compareMeta);
   652→  }
   653→}
